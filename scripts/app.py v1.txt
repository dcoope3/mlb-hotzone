# app.py — auto-update heatmap on selection change (no button)
#
# Behavior:
# - When player / metric / toggles change:
#     1) determine expected PNG path
#     2) if missing OR regenerate enabled
#     3) run Script 5 automatically
#     4) display heatmap
#
# Run:
#   streamlit run app.py

from __future__ import annotations

import sys
import subprocess
from pathlib import Path
import pandas as pd
import streamlit as st

# -----------------------------------------------------
# Project paths
# -----------------------------------------------------
PROJECT_ROOT = Path(__file__).resolve().parent
SCRIPTS_DIR = PROJECT_ROOT / "scripts"

if str(SCRIPTS_DIR) not in sys.path:
    sys.path.insert(0, str(SCRIPTS_DIR))

from config import PROCESSED_DIR, VIZ_DIR  # noqa: E402
from utils_paths import heatmap_path  # noqa: E402

WEIGHTS_PATH = PROCESSED_DIR / "zone_weights_2024.csv"
SCRIPT5_PATH = SCRIPTS_DIR / "05_visualize_hotzones_2024.py"

SEASON = 2024

# -----------------------------------------------------
# Available metrics
# -----------------------------------------------------
METRICS = [
    "zone_weight",
    "xwoba_shrunk",
    "xwoba_contact",
    "bip_count",
    "pitches_seen",
    "babip",
    "iso_bip",
    "contact_rate",
]

# -----------------------------------------------------
# Streamlit settings
# -----------------------------------------------------
st.set_page_config(
    page_title="MLB Hot Zones",
    layout="centered"
)

st.title("MLB Hot Zones (2024)")

st.caption(f"RUNNING: {__file__}")
st.caption(f"METRICS NOW: {METRICS}")
# -----------------------------------------------------
# Helpers
# -----------------------------------------------------
@st.cache_data(show_spinner=False)
def load_players(weights_path: Path) -> pd.DataFrame:

    usecols = ["player_name", "batter"]

    try:
        df = pd.read_csv(weights_path, usecols=usecols, engine="pyarrow")
    except Exception:
        df = pd.read_csv(weights_path, usecols=usecols)

    df = df.dropna(subset=["batter"]).copy()
    df["batter"] = df["batter"].astype(int)

    if "player_name" not in df.columns:
        df["player_name"] = ""

    df["player_name"] = df["player_name"].fillna("").astype(str)

    return (
        df.drop_duplicates(subset=["player_name", "batter"])
        .sort_values("player_name")
        .reset_index(drop=True)
    )


def run_script5(
    batter_id: int,
    metric: str,
    annotate_values: bool,
    annotate_counts: bool,
    overwrite: bool,
):

    cmd = [
        sys.executable,
        str(SCRIPT5_PATH),
        "--batter",
        str(int(batter_id)),
        "--metric",
        metric,
        "--season",
        str(SEASON),
    ]

    if annotate_values:
        cmd.append("--annotate-values")

    if annotate_counts:
        cmd.append("--annotate-counts")

    if overwrite:
        cmd.append("--overwrite")

    p = subprocess.run(
        cmd,
        capture_output=True,
        text=True,
        cwd=str(PROJECT_ROOT),
    )

    return p.returncode, p.stdout, p.stderr


# -----------------------------------------------------
# Guardrails
# -----------------------------------------------------
if not WEIGHTS_PATH.exists():
    st.error(f"Missing weights file: {WEIGHTS_PATH}")
    st.info("Run Script 4 first.")
    st.stop()

if not SCRIPT5_PATH.exists():
    st.error(f"Missing Script 5: {SCRIPT5_PATH}")
    st.stop()

VIZ_DIR.mkdir(parents=True, exist_ok=True)

players = load_players(WEIGHTS_PATH)

# --- Force session-state metric to a valid option (prevents old "batting_avg" lingering) ---
if "metric" not in st.session_state:
    st.session_state.metric = METRICS[0]

if st.session_state.metric not in METRICS:
    st.session_state.metric = METRICS[0]

# -----------------------------------------------------
# Session state
# -----------------------------------------------------
if "last_request_key" not in st.session_state:
    st.session_state.last_request_key = None

if "last_gen_result" not in st.session_state:
    st.session_state.last_gen_result = None

# -----------------------------------------------------
# UI
# -----------------------------------------------------
name_query = st.text_input("Search player:", "")

q = name_query.strip().lower()

if q:
    matches = players[
        players["player_name"].str.lower().str.contains(q, na=False)
    ].head(100)
else:
    matches = players

if matches.empty:
    st.warning("No players found.")
    st.stop()

options = list(matches.itertuples(index=False, name=None))


def format_player(opt):
    return f"{opt[0]} ({int(opt[1])})"


choice = st.selectbox(
    "Player:",
    options=options,
    format_func=format_player
)

player_name = str(choice[0] or "")
batter_id = int(choice[1])

metric = st.selectbox(
    "Metric:",
    METRICS,
    index=METRICS.index(st.session_state.metric),
    key="metric"
)

col1, col2 = st.columns(2)

with col1:
    annotate_values = st.checkbox(
        "Annotate values",
        value=True
    )

with col2:
    annotate_counts = st.checkbox(
        "Annotate BIP counts",
        value=True
    )

col3, col4 = st.columns(2)

with col3:
    auto_generate = st.checkbox(
        "Auto-generate if missing",
        value=True
    )

with col4:
    regenerate_even_if_exists = st.checkbox(
        "Regenerate even if exists",
        value=False
    )

# -----------------------------------------------------
# Image path
# -----------------------------------------------------
image_path = heatmap_path(
    viz_dir=VIZ_DIR,
    season=SEASON,
    metric=metric,
    player_name=player_name,
    batter_id=batter_id,
)

st.caption(f"Expected file: {image_path.name}")

# -----------------------------------------------------
# Auto-generate logic
# -----------------------------------------------------
request_key = (
    batter_id,
    metric,
    annotate_values,
    annotate_counts,
    regenerate_even_if_exists
)

need_generate = False

if regenerate_even_if_exists:
    need_generate = True
else:
    need_generate = not image_path.exists()

selection_changed = request_key != st.session_state.last_request_key

should_run = (
    selection_changed
    and need_generate
    and (auto_generate or regenerate_even_if_exists)
)

if should_run:

    st.session_state.last_request_key = request_key

    with st.spinner("Generating heatmap..."):

        code, out, err = run_script5(
            batter_id=batter_id,
            metric=metric,
            annotate_values=annotate_values,
            annotate_counts=annotate_counts,
            overwrite=regenerate_even_if_exists,
        )

    st.session_state.last_gen_result = (code, out, err)

# -----------------------------------------------------
# Display
# -----------------------------------------------------
if st.session_state.last_gen_result is not None and should_run:

    code, out, err = st.session_state.last_gen_result

    if code != 0:
        st.error("Heatmap generation failed.")
        st.code(err if err else out)

if image_path.exists():

    st.image(
        str(image_path),
        caption=f"{player_name} — {metric}",
        use_container_width=True,
    )

else:

    if auto_generate:
        st.warning(
            "Heatmap file still not found. "
            "Generation may have failed."
        )

    else:
        st.info(
            "Heatmap not found. Turn on Auto-generate "
            "or run Script 6."
        )

        st.code(
            "python scripts/06_precompute_heatmaps_2024.py --only-missing",
            language="bash",
        )
